<script>
function loadAll(){
  const c = window.SITE_CONFIG
  document.getElementById('cfg_title').value = c.base.title
  document.getElementById('cfg_tel').value = c.base.tel
  document.getElementById('cfg_wx').value = c.base.wx
  document.getElementById('cfg_addr').value = c.base.address
  document.getElementById('cfg_swiper').value = JSON.stringify(c.swiper,null,2)
  document.getElementById('cfg_case').value = JSON.stringify(c.cases,null,2)
  document.getElementById('cfg_pkg').value = JSON.stringify(c.packages,null,2)
}
function saveAll(){
  try{
    window.SITE_CONFIG.base.title = document.getElementById('cfg_title').value
    window.SITE_CONFIG.base.tel = document.getElementById('cfg_tel').value
    window.SITE_CONFIG.base.wx = document.getElementById('cfg_wx').value
    window.SITE_CONFIG.base.address = document.getElementById('cfg_addr').value
    window.SITE_CONFIG.swiper = JSON.parse(document.getElementById('cfg_swiper').value)
    window.SITE_CONFIG.cases = JSON.parse(document.getElementById('cfg_case').value)
    window.SITE_CONFIG.packages = JSON.parse(document.getElementById('cfg_pkg').value)
    Store.save()
    alert('保存成功！打开 index.html 刷新查看效果')
  }catch(e){
    alert('JSON格式错误！检查逗号、引号')
    console.error(e)
  }
}
loadAll()
</script>
