// 导航切换
document.querySelectorAll('.nav-item').forEach(item=>{
  item.onclick = ()=>{
    document.querySelectorAll('.nav-item').forEach(i=>i.classList.remove('active'))
    item.classList.add('active')
    const target = item.dataset.page
    document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'))
    document.getElementById(target).classList.add('active')
  }
})

// 渲染页面
function render(){
  const cfg = window.SITE_CONFIG
  document.title = cfg.base.title
  document.getElementById('showTel').innerText = cfg.base.tel
  document.getElementById('showAddr').innerText = cfg.base.address
  document.getElementById('showWx').innerText = cfg.base.wx
  document.getElementById('floatTel').href = `tel:${cfg.base.tel}`

  // 轮播
  let swiperHtml = ''
  cfg.swiper.forEach(s=>{
    swiperHtml += `<div class="swiper-item"><img src="${s.img}"><div class="swiper-text">${s.title}</div></div>`
  })
  document.getElementById('swiperDom').innerHTML = swiperHtml

  // 服务
  let serHtml = ''
  cfg.service.forEach(s=>{
    serHtml += `<div class="service-card"><div class="icon">${s.icon}</div><h4>${s.name}</h4><p>${s.desc}</p></div>`
  })
  document.getElementById('serviceDom').innerHTML = serHtml

  // 优势
  let advHtml = ''
  cfg.advantage.forEach(a=>advHtml += `<span class="adv-tag">${a}</span>`)
  document.getElementById('advDom').innerHTML = advHtml

  // 首页案例横向
  let caseH = ''
  cfg.cases.forEach(c=>{
    caseH += `<div class="case-card"><img src="${c.img}"><div class="case-info"><h4>${c.name}</h4><div class="sub">${c.area}｜${c.style}</div></div></div>`
  })
  document.getElementById('caseHomeDom').innerHTML = caseH

  // 案例全部
  let caseA = ''
  cfg.cases.forEach(c=>{
    caseA += `<div class="case-card" style="margin-bottom:12px;"><img src="${c.img}"><div class="case-info"><h4>${c.name}【${c.tag}】</h4><div class="sub">${c.area}｜${c.style}</div><p style="font-size:12px;color:#666;margin-top:4px;">${c.desc}</p></div></div>`
  })
  document.getElementById('caseAllDom').innerHTML = caseA

  // 套餐
  let pkgH = ''
  cfg.packages.forEach(p=>{
    const inStr = p.include.map(x=>`✅ ${x}`).join('<br>')
    const outStr = p.exclude.map(x=>`❌ ${x}`).join('<br>')
    pkgH += `
      <div class="pkg-item">
        <div class="pkg-top"><h4>${p.name}</h4><div class="pkg-price">${p.price}</div></div>
        <div class="pkg-tip">${p.tip}</div>
        <div class="pkg-list">
          <div class="in">${inStr}</div>
          <div class="out">${outStr}</div>
        </div>
      </div>
    `
  })
  document.getElementById('pkgDom').innerHTML = pkgH
}

// 表单提交（这里是前端模拟，后期对接后端/表单收集）
function submitForm(){
  const data = {
    name:document.getElementById('name').value,
    phone:document.getElementById('phone').value,
    community:document.getElementById('community').value,
    area:document.getElementById('area').value,
    need:document.getElementById('need').value
  }
  if(!data.name || !data.phone){
    alert('请填写姓名和联系电话')
    return
  }
  alert('提交成功！我们设计师会尽快联系您')
}

render()
